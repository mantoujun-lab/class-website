(function() {
    var path = window.location.pathname;
    var isEnglish = /^\/en(\/|$)/.test(path);
    document.documentElement.lang = isEnglish ? 'en' : 'zh-CN';
    var sections = document.querySelectorAll('.error-section');

        } else {
            section.style.display = 'none';
        }
    });
})();
